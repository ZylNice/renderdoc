3. 关键代码段精解

① Albedo 与金属度增益

albedo = texture(_u_MainTex, uv).rgb * _u_BaseColor.rgb; albedoLuma = dot(albedo0, vec3(0.22, 0.707, 0.071)); // 备用（反射提亮） metal = clamp(LightTex.y * MetalRemap.y + MetalRemap.x, 0, 1); albedo *= lerp(_ug_HeroNonMetalAlbedoIntensity, _ug_HeroMetalAlbedoIntensity, metal); albedo += metal * albedo * _u_MetallicAlbedoMultiplier; albedo *= albedo * (albedo * 0.305306 + 0.682171) + 0.012523; // sRGB 快速解码

② 粗糙度与 Matcap

rough = lerp(1.0, clamp(LightTex.x * RoughRemap.y + RoughRemap.x, 0, 1), T2.w); matcap = textureLod(_u_HeroMatCup, T2.xy*0.5+0.5, rough*6.0); matcap.rgb²=转线性

③ 环境高光

F0 = T2.z * clamp(0.532 - rough, 0, 1) + (1 - metal) * 0.04; specRamp = metal * albedo + F0; env = matcap.rgb × HeroEvnIntensity × specRamp × T1.xyz × lerp(NonMetalRef, MetalRef, metal) × (1 + AOIntensity·(LightTex.z−1)) × (albedoLuma^AlbedoLuminancePower × ReflectionBrightenIntensity + 1);

④ 直接光 ⑤ Fresnel ⑥ Glow ⑦ VS 加色

direct = (T1.w × _ug_HeroLightIrradiance) × (matcap.a × specRamp × T2.w); fresnel = (UseFullFresnel ? 1−LightTex.z : LightTex.z) × T3.w × _u_FresnelColor.rgb; glow = LightTex.w × _u_GlowColor² × _u_GlowIntension; color += clamp(T4.xyz)·_u_MulColor + clamp(T3.xyz); // 受击染色 / Rim·Hurt

⑧ 亮度雾（保饱和）

diff = luma(_ug_HeroFogColor.rgb) − luma(color); // 权重 (0.3, 0.59, 0.11) fogged = color + diff × _uvs_TEXCOORD7; // 只推亮度不动色相 minC<0 → 沿亮度方向回拉；maxC>1 → 向亮度压缩 // 保饱和修正

⑨ 曝光 / HDR / Tonemap / 输出

color ×= (_ug_ExposureScale.rgb + 1) × _u_ExposureColor; if (_ug_ExposureScale.w > 0) HDR 分支（w>1 不压，否则 ×0.5），a=PlayerId; ldr = c·(0.533c+0.151)/(c·(0.526c+0.200)+0.027); a=PlayerId; // Filmic out = HDR ? hdr : ldr; // 本帧"有泛光"：HDR 分支输出供 Bloom 阈值提取亮斑
本截帧为有泛光版本：英雄 HDR 输出经后续 Bloom 链（EID 7063 起，阈值 _u_Threshold 提取）产生辉光。若英雄高光/Glow 过曝导致泛光糊一片，排查顺序：先看 _ug_ExposureScale.w 是否走 HDR 分支 → 再看 Bloom 阈值 → 最后才是材质本身。
